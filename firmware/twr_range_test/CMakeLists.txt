cmake_minimum_required(VERSION 3.20.0)

set(IMEC_APP_DIR "${CMAKE_CURRENT_LIST_DIR}/../app")
list(APPEND DTS_ROOT "${IMEC_APP_DIR}")

set(TWR_RANGE_ROLE "initiator" CACHE STRING "DS-TWR range-test role: initiator or responder")
set_property(CACHE TWR_RANGE_ROLE PROPERTY STRINGS initiator responder)

set(TWR_INITIATOR_ID "0x1111222233334444ull" CACHE STRING "DS-TWR initiator device ID")
set(TWR_RESPONDER_ID "0x5555666677778888ull" CACHE STRING "DS-TWR responder device ID")
set(TWR_NETWORK_ID "0x1A2B3C4Du" CACHE STRING "DS-TWR test network ID")
set(TWR_SESSION_ID "0x0000A11Eu" CACHE STRING "DS-TWR test session ID")
set(TWR_SESSION_NONCE "0x1020304050607080ull" CACHE STRING "DS-TWR test session nonce")
set(TWR_REPLY_DELAY_UUS "20000u" CACHE STRING "DS-TWR test delayed-TX reply delay in DW1000/DW3000 UWB microseconds")
set(TWR_RX_AFTER_TX_DELAY_UUS "0u" CACHE STRING "DS-TWR test RX-after-TX delay in DW1000/DW3000 UWB microseconds")

set(TWR_RANGE_PERIOD_MS "1000u" CACHE STRING "Delay between initiator range attempts in milliseconds")
set(TWR_LED_FLASH "1" CACHE STRING "Flash board LEDs for range-attempt status")
set(TWR_CAPTURE_RSL "1" CACHE STRING "Capture RSL/CIR diagnostics where the range-test driver supports it")
set(TWR_CLICKER_DIAG "1" CACHE STRING "Send the post-range clicker diagnostic frame")
set(TWR_LOG_EACH_RANGE "1" CACHE STRING "Log every range attempt, not only calibration summaries")

set(TWR_PHY_PREAMBLE_LENGTH "DWT_PLEN_4096" CACHE STRING "DS-TWR test preamble length")
set(TWR_PHY_RX_PAC "DWT_PAC32" CACHE STRING "DS-TWR test RX preamble acquisition chunk")
set(TWR_PHY_TX_CODE "9" CACHE STRING "DS-TWR test TX preamble code")
set(TWR_PHY_RX_CODE "9" CACHE STRING "DS-TWR test RX preamble code")

set(TWR_TX_POWER "0xffffffffu" CACHE STRING "DS-TWR test TX power register value")
set(TWR_TX_ANT_DLY "16385u" CACHE STRING "DS-TWR test TX antenna delay")
set(TWR_RX_ANT_DLY "16385u" CACHE STRING "DS-TWR test RX antenna delay")

if(TWR_RANGE_ROLE STREQUAL "initiator")
    set(TWR_DEVICE_ID "${TWR_INITIATOR_ID}")
elseif(TWR_RANGE_ROLE STREQUAL "responder")
    set(TWR_DEVICE_ID "${TWR_RESPONDER_ID}")
else()
    message(FATAL_ERROR "TWR_RANGE_ROLE must be initiator or responder")
endif()

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(imec_twr_range_test)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/dwm3000_sdk.cmake")
imec_resolve_dwm3000_sdk(DWM3000_SDK_DIR)

target_sources(app PRIVATE
    src/main.c
    "${IMEC_APP_DIR}/src/dwm3000_driver.c"
    "${IMEC_APP_DIR}/src/dwm3000_port.c"
    "${IMEC_APP_DIR}/src/dwm3000_sdk_port.c"
    "${DWM3000_SDK_DIR}/decadriver/deca_device.c"
    ../src/debug_log.c
    ../src/protocol.c
    ../src/uwb.c
    ../src/uwb_session.c
)

target_include_directories(app PRIVATE
    ../include
    "${IMEC_APP_DIR}/src"
    "${DWM3000_SDK_DIR}/decadriver"
)

target_compile_definitions(app PRIVATE
    DEVICE_ID=${TWR_DEVICE_ID}
    TWR_INITIATOR_ID=${TWR_INITIATOR_ID}
    TWR_RESPONDER_ID=${TWR_RESPONDER_ID}
    TWR_NETWORK_ID=${TWR_NETWORK_ID}
    TWR_SESSION_ID=${TWR_SESSION_ID}
    TWR_SESSION_NONCE=${TWR_SESSION_NONCE}
    DWM3000_DS_TWR_REPLY_DLY_UUS=${TWR_REPLY_DELAY_UUS}
    DWM3000_POLL_TX_TO_RESP_RX_DLY_UUS=${TWR_RX_AFTER_TX_DELAY_UUS}
    DWM3000_RESP_TX_TO_FINAL_RX_DLY_UUS=${TWR_RX_AFTER_TX_DELAY_UUS}
    DWM3000_RESP_RX_TIMEOUT_UUS=${TWR_RX_TIMEOUT_UUS}
    DWM3000_FINAL_RX_TIMEOUT_UUS=${TWR_RX_TIMEOUT_UUS}
    DWM3000_REPORT_RX_TIMEOUT_UUS=${TWR_RX_TIMEOUT_UUS}
    DWM3000_RESP_RX_TIMEOUT_MS=${TWR_RX_TIMEOUT_MS}
    DWM3000_FINAL_RX_TIMEOUT_MS=${TWR_RX_TIMEOUT_MS}
    DWM3000_REPORT_RX_TIMEOUT_MS=${TWR_RX_TIMEOUT_MS}
    DWM3000_REPLY_TIMING_TOLERANCE_US=${TWR_REPLY_TOLERANCE_US}
    DWM3000_REPLY_DELAY_FROM_ROUND_INDEX=${TWR_DELAY_CALIBRATION}
    DWM3000_REPLY_DELAY_CALIBRATION_MAX_UUS=${TWR_REPLY_DELAY_UUS}
    DWM3000_REPLY_DELAY_CALIBRATION_MIN_UUS=${TWR_REPLY_DELAY_MIN_UUS}
    DWM3000_REPLY_DELAY_CALIBRATION_STEP_UUS=${TWR_REPLY_DELAY_STEP_UUS}
    DWM3000_CLICKER_DIAG_ENABLED=${TWR_CLICKER_DIAG}
    DWM3000_CIA_DIAG_MODE=${TWR_CIA_DIAG_MODE}
    DWM3000_PHY_CHANNEL=${TWR_PHY_CHANNEL}
    DWM3000_PHY_PREAMBLE_LENGTH=${TWR_PHY_PREAMBLE_LENGTH}
    DWM3000_PHY_RX_PAC=${TWR_PHY_RX_PAC}
    DWM3000_PHY_TX_CODE=${TWR_PHY_TX_CODE}
    DWM3000_PHY_RX_CODE=${TWR_PHY_RX_CODE}
    DWM3000_PHY_SFD_TYPE=${TWR_PHY_SFD_TYPE}
    DWM3000_PHY_DATA_RATE=${TWR_PHY_DATA_RATE}
    DWM3000_PHY_SFD_TIMEOUT=${TWR_PHY_SFD_TIMEOUT}
    DWM3000_TX_PG_DELAY=${TWR_TX_PG_DELAY}
    DWM3000_TX_POWER=${TWR_TX_POWER}
    DWM3000_TX_ANT_DLY=${TWR_TX_ANT_DLY}
    DWM3000_RX_ANT_DLY=${TWR_RX_ANT_DLY}
    TWR_DELAY_CALIBRATION=${TWR_DELAY_CALIBRATION}
    TWR_REPLY_DELAY_CALIBRATION_MAX_UUS=${TWR_REPLY_DELAY_UUS}
    TWR_REPLY_DELAY_CALIBRATION_MIN_UUS=${TWR_REPLY_DELAY_MIN_UUS}
    TWR_REPLY_DELAY_CALIBRATION_STEP_UUS=${TWR_REPLY_DELAY_STEP_UUS}
    TWR_REPLY_DELAY_INITIAL_STEP_UUS=${TWR_REPLY_DELAY_INITIAL_STEP_UUS}
    TWR_REPLY_DELAY_TESTS_PER_STEP=${TWR_REPLY_DELAY_TESTS_PER_STEP}
    TWR_REPLY_DELAY_REQUIRED_SUCCESSES=${TWR_REPLY_DELAY_REQUIRED_SUCCESSES}
    TWR_RANGE_PERIOD_MS=${TWR_RANGE_PERIOD_MS}
    TWR_LED_FLASH=${TWR_LED_FLASH}
    TWR_CAPTURE_RSL=${TWR_CAPTURE_RSL}
    TWR_CLICKER_DIAG=${TWR_CLICKER_DIAG}
    TWR_LOG_EACH_RANGE=${TWR_LOG_EACH_RANGE}
    IMEC_BUILD_PRESET_NAME="twr_range_${TWR_RANGE_ROLE}"
    IMEC_GIT_VERSION="custom"
    IMEC_BUILD_TIMESTAMP="custom"
    IMEC_BOARD_TARGET="${BOARD}"
)

if(TWR_RANGE_ROLE STREQUAL "initiator")
    target_compile_definitions(app PRIVATE TWR_RANGE_INITIATOR=1 DEVICE_ROLE=1)
else()
    target_compile_definitions(app PRIVATE TWR_RANGE_RESPONDER=1 DEVICE_ROLE=2)
