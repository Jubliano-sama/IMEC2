    "Power profile mode: clicker_sleep, clicker_systemoff, clicker_systemoff_nowake, anchor_sleep, anchor_low_duty, clicker_click, or responder")
set_property(CACHE POWER_PROFILE_MODE PROPERTY STRINGS
    clicker_sleep clicker_systemoff clicker_systemoff_nowake anchor_sleep anchor_low_duty clicker_click responder)

if(POWER_PROFILE_MODE STREQUAL "clicker_sleep")
    set(POWER_PROFILE_DEVICE_ID "${POWER_PROFILE_CLICKER_ID}")
    set(POWER_PROFILE_DEVICE_ROLE "1")
    set(POWER_PROFILE_MODE_DEFINE "POWER_PROFILE_MODE_CLICKER_SLEEP=1")
elseif(POWER_PROFILE_MODE STREQUAL "clicker_systemoff")
    set(POWER_PROFILE_DEVICE_ID "${POWER_PROFILE_CLICKER_ID}")
    set(POWER_PROFILE_DEVICE_ROLE "1")
    set(POWER_PROFILE_MODE_DEFINE "POWER_PROFILE_MODE_CLICKER_SYSTEMOFF=1")
elseif(POWER_PROFILE_MODE STREQUAL "clicker_systemoff_nowake")
    set(POWER_PROFILE_DEVICE_ID "${POWER_PROFILE_CLICKER_ID}")
    set(POWER_PROFILE_DEVICE_ROLE "1")
    set(POWER_PROFILE_MODE_DEFINE "POWER_PROFILE_MODE_CLICKER_SYSTEMOFF_NOWAKE=1")
elseif(POWER_PROFILE_MODE STREQUAL "anchor_sleep")
    set(POWER_PROFILE_DEVICE_ID "${POWER_PROFILE_RESPONDER0_ID}")
    set(POWER_PROFILE_DEVICE_ROLE "2")
    set(POWER_PROFILE_MODE_DEFINE "POWER_PROFILE_MODE_ANCHOR_SLEEP=1")
elseif(POWER_PROFILE_MODE STREQUAL "anchor_low_duty")
    set(POWER_PROFILE_DEVICE_ID "${POWER_PROFILE_RESPONDER0_ID}")
    set(POWER_PROFILE_DEVICE_ROLE "2")
    set(POWER_PROFILE_MODE_DEFINE "POWER_PROFILE_MODE_ANCHOR_LOW_DUTY=1")
elseif(POWER_PROFILE_MODE STREQUAL "clicker_click")
    set(POWER_PROFILE_DEVICE_ID "${POWER_PROFILE_CLICKER_ID}")
    set(POWER_PROFILE_DEVICE_ROLE "1")
    set(POWER_PROFILE_MODE_DEFINE "POWER_PROFILE_MODE_CLICKER_CLICK=1")
elseif(POWER_PROFILE_MODE STREQUAL "responder")
    set(POWER_PROFILE_DEVICE_ID "${POWER_PROFILE_RESPONDER0_ID}")
    set(POWER_PROFILE_DEVICE_ROLE "2")
    set(POWER_PROFILE_MODE_DEFINE "POWER_PROFILE_MODE_RESPONDER=1")
else()
    message(FATAL_ERROR "Unsupported POWER_PROFILE_MODE=${POWER_PROFILE_MODE}")
endif()
